function seeded(seed: number) {
    // Lo stato di partenza è un hash del seme: semi consecutivi non danno personaggi correlati.
    let a = Math.imul(seed + 0x9e3779b9, 0x85ebca6b) ^ 0xc2b2ae35;
    const next = () => {
        a = (a + 0x6d2b79f5) | 0;
        let t = Math.imul(a ^ (a >>> 15), 1 | a);
        t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
        return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    };
    for (let i = 0; i < 32; i++) next();
    return next;
}

type Beat = { step: number; day: number; hour: number; kind: string; id: string; extra?: unknown };

const DIRS: Array<[number, number]> = [[1, 0], [0, 1], [-1, 0], [0, -1]];

/** Dijkstra over the map: forest and water cost more time. */
function route(map: string[], from: { x: number; y: number }, isGoal: (x: number, y: number) => boolean, avoid: Set<string>) {
    const cost = (ch: string) => (ch === 'F' ? 2 : ch === '~' ? 3 : 1);
    const H = map.length, W = map[0].length;
    const dist = new Map<number, number>();
    const prev = new Map<number, number>();
    const key = (x: number, y: number) => y * W + x;
    const queue: Array<[number, number, number]> = [[0, from.x, from.y]];
    dist.set(key(from.x, from.y), 0);
    while (queue.length) {
        queue.sort((p, q) => p[0] - q[0]);
        const [d, x, y] = queue.shift()!;
        if (d > (dist.get(key(x, y)) ?? Infinity)) continue;
        if ((x !== from.x || y !== from.y) && isGoal(x, y)) {
            let k = key(x, y);
            const path: Array<[number, number]> = [];
            while (k !== key(from.x, from.y)) { path.push([k % W, Math.floor(k / W)]); k = prev.get(k)!; }
            return path.reverse();
        }
        for (const [dx, dy] of DIRS) {
            const nx = x + dx, ny = y + dy;
            if (nx < 0 || ny < 0 || nx >= W || ny >= H) continue;
            const ch = map[ny][nx];
            if (ch === 'M' || avoid.has(`${nx},${ny}`)) continue;
            const nd = d + cost(ch);
            if (nd < (dist.get(key(nx, ny)) ?? Infinity)) { dist.set(key(nx, ny), nd); prev.set(key(nx, ny), key(x, y)); queue.push([nd, nx, ny]); }
        }
    }
    return null;
}

function equipBest() {
    const c = character();
    const items = useItemDatabaseStore.getState().itemDatabase;
    let bestW = -1, wScore = c.equippedWeapon !== null ? score(c.inventory[c.equippedWeapon]?.itemId) : 0;
    function score(id?: string) {
        const d = id ? items[id] : undefined;
        if (!d) return 0;
        if (d.type === 'weapon') {
            if (d.weaponType === 'thrown') return 0;
            if (d.weaponType === 'ranged' && (!d.ammoType || c.getItemCount(d.ammoType) <= 0)) return 0;
            return (d.damage ?? 0);
        }
        return d.defense ?? 0;
    }
    c.inventory.forEach((e, i) => {
        const d = items[e.itemId];
        if (!d || d.type !== 'weapon' || (e.durability && e.durability.current <= 0)) return;
        const s = score(e.itemId);
        if (s > wScore) { wScore = s; bestW = i; }
    });
    if (bestW >= 0) c.equipItem(bestW);
    for (const slot of ['chest', 'head', 'legs'] as const) {
        const cur = slot === 'chest' ? c.equippedArmor : slot === 'head' ? c.equippedHead : c.equippedLegs;
        let best = -1, bs = cur !== null ? score(character().inventory[cur]?.itemId) : 0;
        character().inventory.forEach((e, i) => {
            const d = items[e.itemId];
            if (!d || d.type !== 'armor' || (d.slot ?? 'chest') !== slot || (e.durability && e.durability.current <= 0)) return;
            const s = score(e.itemId);
            if (s > bs) { bs = s; best = i; }
        });
        if (best >= 0) character().equipItem(best);
    }
}

function useBest(effect: string, preferLarge: boolean) {
    const c = character();
    const items = useItemDatabaseStore.getState().itemDatabase;
    let idx = -1, bestVal = preferLarge ? -1 : 1e9;
    c.inventory.forEach((e, i) => {
        const d = items[e.itemId];
        if (d?.type !== 'consumable') return;
        const f = d.effects?.find(x => x.type === effect);
        if (!f || typeof f.value !== 'number' || f.value <= 0) return;
        if (preferLarge ? f.value > bestVal : f.value < bestVal) { bestVal = f.value; idx = i; }
    });
    return idx !== -1 && applyItemUse(idx) ? true : false;
}

const careful = process.env.BOT !== 'naive';
const diligent = process.env.BOT === 'diligent';
function runJourney(seed: number) {
    const random = seeded(seed);
    vi.spyOn(Math, 'random').mockImplementation(random);
    const pick = <T,>(l: T[]): T => l[Math.floor(random() * l.length)];
    const roll = () => { const r = [1, 2, 3, 4].map(() => 1 + Math.floor(random() * 6)).sort((a, b) => a - b); return r[1] + r[2] + r[3]; };

    startNewGame();
    narrativeService.skipCutscene();
    character().setAttributes({ for: roll(), des: roll(), cos: roll(), int: roll(), sag: roll(), car: roll() });
    const attrs0 = { ...character().attributes };
    game().setGameState(GameState.IN_GAME);

    const beats: Beat[] = [];
    const rec = (kind: string, id: string, extra?: unknown) => {
        const t = useTimeStore.getState().gameTime;
        beats.push({ step: game().totalSteps, day: t.day, hour: t.hour, kind, id, extra });
    };
    const stat = { combats: 0, wins: 0, flees: 0, dmgTaken: 0, healsUsed: 0, foodUsed: 0, waterUsed: 0, quickRests: 0, refugeSleeps: 0, stuckEnd: 0, hpMin: 1e9, events: 0 };
    const seenQ = new Set<string>(); const seenC = new Set<string>();
    let lastState = -1, lastStage = game().mainStoryStage, lastEvent = '', lastCombatId = -1;
    const visitedRefuges = new Set<string>();
    const samples: number[][] = []; const seenStatus = new Set<string>(); let lastSample = -1; let combatDmg = 0;
    let lastSearchBiome = '';
    let outcome = 'timeout';
    let detour: Array<[number, number]> | null = null;

    for (let action = 0; action < 12000; action++) {
        const g = game(); const c = character();
        stat.hpMin = Math.min(stat.hpMin, c.hp.current);
        if (g.gameState === GameState.VICTORY) { outcome = 'victory'; break; }
        if (g.gameState === GameState.GAME_OVER) { outcome = `death:${g.deathCause}`; break; }

        const cnt = (eff: string) => { const items = useItemDatabaseStore.getState().itemDatabase; let n = 0; for (const e of c.inventory) { const d = items[e.itemId]; if (d?.effects?.some(f => f.type === eff && typeof f.value === 'number' && f.value > 0)) n += e.quantity; } return n; };
        if (g.totalSteps % 10 === 0 && g.totalSteps !== lastSample) { lastSample = g.totalSteps; const t = useTimeStore.getState().gameTime; samples.push([g.totalSteps, t.day, t.hour, c.hp.current, c.hp.max, Math.round(c.satiety.current), Math.round(c.hydration.current), Math.round(c.fatigue.current), cnt('satiety'), cnt('hydration'), cnt('heal'), c.level]); }
        for (const st of c.status) if (!seenStatus.has(st + game().totalSteps)) { seenStatus.add(st + game().totalSteps); rec('status', st); }
        // --- record transitions ---
        for (const q of Object.keys(c.activeQuests)) if (!seenQ.has('a' + q)) { seenQ.add('a' + q); rec('questStart', q); }
        for (const q of c.completedQuests) if (!seenQ.has('c' + q)) { seenQ.add('c' + q); rec('questDone', q); }
        if (g.gameState !== lastState) {
            if (g.gameState === GameState.MAIN_STORY && g.activeMainStoryEvent) rec('chapter', String(g.activeMainStoryEvent.stage));
            if (g.gameState === GameState.CUTSCENE) rec('cutscene', g.activeCutscene?.id ?? 'ink');
            lastState = g.gameState;
        }
        const ae = useEventStore.getState().activeEvent;
        if (g.gameState === GameState.EVENT_SCREEN && ae && ae.id !== lastEvent) { lastEvent = ae.id; stat.events++; rec('event', ae.id); }
        const ac = useCombatStore.getState().activeCombat;
        if (g.gameState === GameState.COMBAT && ac && useCombatStore.getState().combatId !== lastCombatId) {
            lastCombatId = useCombatStore.getState().combatId; stat.combats++; rec('combat', ac.enemy.id, { hp: c.hp.current, lvl: c.level });
        }

        // --- act ---
        if (c.levelUpPending && g.gameState === GameState.IN_GAME) {
            const talents = useTalentDatabaseStore.getState().talents.filter(t => c.level + 1 >= t.levelRequirement && c.skills[t.requiredSkill]?.proficient && !c.unlockedTalents.includes(t.id));
            const learnable = (Object.keys(SKILLS) as SkillName[]).filter(s => !c.skills[s]?.proficient);
            c.applyLevelUp({ attribute: pick(['for', 'cos', 'des'] as const), talentId: talents.length ? pick(talents).id : undefined, proficiency: !talents.length && learnable.length ? pick(learnable) : undefined });
            rec('levelUp', String(character().level));
            continue;
        }
        switch (g.gameState) {
            case GameState.IN_GAME: {
                const inter = useInteractionStore.getState();
                if (inter.isInRefuge) {
                    const { options } = inter.refugeMenuState;
                    const hour = useTimeStore.getState().gameTime.hour;
                    const night = hour >= 20 || hour < 6;
                    if (diligent && craftingService.canCraft('recipe_purify_water')) { craftingService.craft('recipe_purify_water'); break; }
                    // A careful player eats and drinks before sleeping: otherwise the night makes them sick.
                    if (careful && options.includes(REFUGE_ACTION.SLEEP)) {
                        const mins = (hour < 6 ? 6 - hour : 24 - hour + 6) * 60;
                        const { satietyCost, hydrationCost } = c.calculateSurvivalCost(mins);
                        if (c.hydration.current < hydrationCost + 1 && useBest('hydration', false)) { stat.waterUsed++; break; }
                        if (c.satiety.current < satietyCost + 1 && useBest('satiety', false)) { stat.foodUsed++; break; }
                    }
                    let choice: string = REFUGE_ACTION.LEAVE;
                    const mins0 = (hour < 6 ? 6 - hour : 24 - hour + 6) * 60; const cs0 = c.calculateSurvivalCost(mins0);
                    const fedNow = c.hydration.current >= cs0.hydrationCost && c.satiety.current >= cs0.satietyCost;
                    if (options.includes(REFUGE_ACTION.SLEEP) && !(process.env.SIM_WARNED === '1' && !fedNow) && (c.fatigue.current > 35 || c.hp.current < c.hp.max * 0.7 || night)) { choice = REFUGE_ACTION.SLEEP; stat.refugeSleeps++; const mins2 = (hour < 6 ? 6 - hour : 24 - hour + 6) * 60; const cs = c.calculateSurvivalCost(mins2); rec('sleep', c.hydration.current >= cs.hydrationCost && c.satiety.current >= cs.satietyCost ? 'fed' : 'UNFED', { hyd: Math.round(c.hydration.current), hydCost: Math.round(cs.hydrationCost), sat: Math.round(c.satiety.current), satCost: Math.round(cs.satietyCost), hour }); }
                    else if (options.includes(REFUGE_ACTION.SEARCH)) choice = REFUGE_ACTION.SEARCH;
                    else if (options.includes(REFUGE_ACTION.WAIT) && c.hp.current < c.hp.max * 0.6) choice = REFUGE_ACTION.WAIT;
                    useInteractionStore.setState(s => ({ refugeMenuState: { ...s.refugeMenuState, selectedIndex: options.indexOf(choice as never) } }));
                    useInteractionStore.getState().confirmRefugeMenuSelection();
                    break;
                }
                if (action % 6 === 0) equipBest();
                if (careful && (c.status.has('MALATO') || c.status.has('AVVELENATO') || c.status.has('INFEZIONE'))) {
                    const items = useItemDatabaseStore.getState().itemDatabase;
                    const idx = c.inventory.findIndex(e => items[e.itemId]?.effects?.some(f => f.type === 'cureStatus' && c.status.has(f.value as never)));
                    if (idx !== -1 && applyItemUse(idx)) { stat.healsUsed++; break; }
                }
                if (c.hydration.current < 40 && useBest('hydration', false)) { stat.waterUsed++; break; }
                if (c.satiety.current < 40 && useBest('satiety', false)) { stat.foodUsed++; break; }
                if (c.hp.current < c.hp.max * 0.5) {
                    if (useBest('heal', true)) { stat.healsUsed++; break; }
                    const before = game().lastRestTime; g.performQuickRest();
                    if (game().lastRestTime !== before) { stat.quickRests++; break; }
                }
                if (c.fatigue.current > 90) { const b = game().lastRestTime; g.performQuickRest(); if (game().lastRestTime !== b) { stat.quickRests++; break; } }

                if (diligent && g.currentBiome !== lastSearchBiome && g.currentBiome !== '~') { lastSearchBiome = g.currentBiome; g.performActiveSearch(); break; }
                const map = g.map; const here = g.playerPos;
                const hour = useTimeStore.getState().gameTime.hour;
                const night = hour >= 20 || hour < 6;
                const tired = c.fatigue.current > 60 || night || c.hp.current < c.hp.max * 0.5;
                const avoid = new Set([...g.visitedRefuges.map(p => `${p.x},${p.y}`)]);
                avoid.delete(`${here.x},${here.y}`);
                // Head for a nearby unused refuge to sleep, otherwise for the Safe Place.
                let step: [number, number] | undefined;
                if (tired) {
                    const toR = route(map, here, (x, y) => map[y][x] === 'R' && !avoid.has(`${x},${y}`), avoid);
                    if (toR && toR.length <= 14) step = toR[0];
                }
                const onEnd = map[here.y][here.x] === 'E';
                if (!step && onEnd) {
                    stat.stuckEnd++;
                    const b = game().lastRestTime; g.performQuickRest();
                    if (game().lastRestTime === b) useTimeStore.getState().advanceTime(120, true);
                    gameService.movePlayer(0, 0);
                    g.checkMainStoryTriggers();
                    break;
                }
                if (!step) {
                    const toE = route(map, here, (x, y) => map[y][x] === 'E', avoid);
                    step = toE ? toE[0] : [here.x + 1, here.y];
                }
                gameService.movePlayer(step[0] - here.x, step[1] - here.y);
                break;
            }
            case GameState.EVENT_SCREEN: {
                const ev = useEventStore.getState();
                if (ev.eventResolutionText || !ev.activeEvent) { ev.dismissEventResolution(); break; }
                const owned = (id: string) => character().getItemCount(id);
                const usable = ev.activeEvent.choices.map((choice, index) => ({ choice, index }))
                    .filter(({ choice }) => ev.isChoiceVisible(choice) && (choice.itemRequirements ?? []).every(r => owned(r.itemId) >= r.quantity));
                if (!usable.length) { ev.dismissEventResolution(); break; }
                ev.resolveEventChoice(pick(usable).index);
                break;
            }
            case GameState.COMBAT: {
                const cb = useCombatStore.getState().activeCombat;
                if (!cb) break;
                if (cb.victory) { combatDmg += cb.damageTaken ?? 0; stat.dmgTaken = combatDmg; useCombatStore.getState().cleanupCombat(); stat.wins++; break; }
                if (cb.playerTurn) {
                    const low = c.hp.current < c.hp.max * 0.35;
                    if (low && useBest('heal', true)) stat.healsUsed++;
                    else if (low && random() < 0.6) { useCombatStore.getState().playerCombatAction({ type: 'flee' }); if (!useCombatStore.getState().activeCombat) stat.flees++; }
                    else useCombatStore.getState().playerCombatAction({ type: 'attack' });
                }
                vi.advanceTimersByTime(ENEMY_TURN_DELAY + 10);
                break;
            }
            case GameState.DIALOGUE:
            case GameState.CUTSCENE: {
                const n = useNarrativeStore.getState();
                if (n.isStoryActive) {
                    if (n.currentChoices.length) narrativeService.chooseChoiceIndex(pick(n.currentChoices).index);
                    else if (g.gameState === GameState.CUTSCENE) narrativeService.skipCutscene(); else narrativeService.endDialogue();
                } else if (g.activeCutscene) g.endCutscene(); else g.setGameState(GameState.IN_GAME);
                break;
            }
            case GameState.MAIN_STORY: g.resolveMainStory(); break;
            case GameState.ASH_LULLABY_CHOICE: g.setGameState(GameState.IN_GAME); useInteractionStore.getState().confirmRefugeMenuSelection(); break;
            case GameState.OUTPOST: g.setGameState(GameState.IN_GAME); break;
            case GameState.TRADING: tradingService.cancelTrade(); break;
            default: g.setGameState(GameState.IN_GAME);
        }
        vi.advanceTimersByTime(1);
    }
    const t = useTimeStore.getState().gameTime;
    const c = character();
    const journalTail = game().journal.slice(0, 8).map(j => j.text);
    return { seed, attrs0, outcome, samples, journalTail, supplies: { food: 0 }, steps: game().totalSteps, day: t.day, hour: t.hour, level: c.level, xp: c.xp.current, hp: c.hp.max, stat, beats, wins: game().totalCombatWins, weapon: c.equippedWeapon !== null ? c.inventory[c.equippedWeapon]?.itemId : null, attrs: c.attributes };
}

describe('balance analysis', () => {
    it('journeys', async () => {
        await newGame();
        const db = useItemDatabaseStore.getState().itemDatabase as Record<string, any>;
        if (process.env.SIM_BOTTLE) db['CONS_002'].effects[0].value = Number(process.env.SIM_BOTTLE);
        if (process.env.SIM_KNIFE_DUR) db['weapon_knife_rusty'].durability = Number(process.env.SIM_KNIFE_DUR);
        const runs = Number(process.env.BALANCE_RUNS ?? 30);
        const out: unknown[] = [];
        for (let seed = Number(process.env.BALANCE_FROM ?? 1); seed < Number(process.env.BALANCE_FROM ?? 1) + runs; seed++) {
            vi.useFakeTimers();
            out.push(runJourney(seed));
            vi.useRealTimers(); vi.restoreAllMocks();
            vi.spyOn(console, 'log').mockImplementation(() => {});
        }
        fs.writeFileSync(process.env.BALANCE_OUT!, JSON.stringify(out));
    }, 600_000);
});
